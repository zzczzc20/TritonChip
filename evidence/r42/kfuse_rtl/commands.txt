#   per vector file: python3 tools/r42/kfuse_rtl_vectors.py --G G --W W --N N --seed SEED --procs 4 --out vec_G<G>_W<W>_N<N>_s<SEED>.hex
#   per run: iverilog -g2005 -Ptb_kfuse_lane.G=.. -Ptb_kfuse_lane.W=.. tools/r42/kfuse_rtl/tb_kfuse_lane.v rtl/core/tnca_kfuse_lane.v \

python3 tools/r42/kfuse_rtl_vectors.py --G 4 --W 0 --N 100000 --seed 21 --procs 1 --out mutation/vec_G4_W0_N100000_s21.hex
python3 tools/r42/kfuse_rtl/mutation_check.py mutation/vec_G4_W0_N100000_s21.hex 4 0 mutation/work

python3 tools/r42/kfuse_rtl/result_classes.py <G> vec_G<G>_W0_N1000000_s1.hex vec_G<G>_W3_N1000000_s1.hex
# post-synthesis netlist simulation (xsim needs /usr/bin/gcc, absent on the x86 host): netlists from the x86 host (below), then on GB10
iverilog -g2012 -DXIL_ENABLE_GLBL -s tb_kfuse_lane -s glbl -DNETLIST -Ptb_kfuse_lane.G=<G> -Ptb_kfuse_lane.W=<W> -Ptb_kfuse_lane.NMAX=30000 \
    tools/r42/kfuse_rtl/tb_kfuse_lane.v netlist_<n>.v unisim/*.v          # unisim/: CARRY8 DSP48E2 FDRE GND VCC LUT1-6 LUT6_2 MUXF7 MUXF8 glbl from Vivado 2023.2 data/verilog/src
vvp -n <n>_unisim.vvp +VEC=nl_<G>_<W>.hex +N=20000 +LAT=<LAT> +LATF=<LAT_FRONT> +MAXERR=20
#   nl_*.hex = first 5,000 lines + last 15,000 lines of a 24,000-line extract (first 6,000 + last 18,000 lines) of the 1,000,000-vector file of that (G, W)
